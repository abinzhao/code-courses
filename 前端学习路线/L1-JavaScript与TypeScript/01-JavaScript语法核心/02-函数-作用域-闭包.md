## 内容讲解

### 1. 函数的三种定义方式

#### 1.1 函数声明（Function Declaration）

```js
// 函数声明会被整体提升（声明 + 定义一起提升）
sayHello(); // ✅ 可以在定义前调用！

function sayHello() {
  console.log('Hello!');
}
```

#### 1.2 函数表达式（Function Expression）

```js
// 函数表达式：把函数赋值给变量
// 变量名被提升，但赋值不提升
greet(); // ❌ TypeError: greet is not a function

var greet = function() {
  console.log('Hi!');
};

// 用 const/let 更好
const sayBye = function() {
  console.log('Bye!');
};
```

#### 1.3 箭头函数（Arrow Function）

```js
// 简洁写法
const add = (a, b) => a + b;
const double = n => n * 2;        // 只有一个参数时可省略括号
const greet = () => 'Hello';      // 无参数时必须写空括号
const getObj = () => ({ x: 1 });  // 返回对象时加括号，避免被当成函数体

// 多行函数体用 {}
const compute = (a, b) => {
  const result = a + b;
  return result * 10;
};
```

#### 1.4 三种方式的 this 差异（关键！）

```js
const obj = {
  name: 'Alice',
  fn1: function() {
    console.log(this.name); // this → 调用者 obj
  },
  fn2: () => {
    console.log(this.name); // this → 外层作用域（此处是全局/window）
  }
};

obj.fn1(); // 'Alice' —— this 是 obj
obj.fn2(); // undefined —— 箭头函数继承外层 this，不是 obj

// 本质：箭头函数没有自己的 this，它的 this 在定义时就决定了
```

> **避坑提示**：对象方法用普通函数，回调中用箭头函数可避免 `this` 丢失。

### 2. 作用域（Scope）

#### 2.1 作用域链

JavaScript 在查找变量时，从当前作用域开始，依次向外层作用域查找，直到全局作用域。如果全局也没找到，抛出 `ReferenceError`。

```js
const globalVar = 'global';

function outer() {
  const outerVar = 'outer';

  function inner() {
    const innerVar = 'inner';
    console.log(innerVar);  // 当前作用域找到
    console.log(outerVar);  // 向外查找到 outer
    console.log(globalVar); // 再向外查找到 global
  }

  inner();
  console.log(innerVar);    // ❌ ReferenceError：无法向内查找
}

outer();
```

#### 2.2 三种作用域

```js
// 1. 全局作用域
const global = '在任何地方都能访问我';

function demo() {
  // 2. 函数作用域
  const funcScope = '只在 demo 函数内可访问';

  for (let i = 0; i < 1; i++) {
    // 3. 块级作用域（let/const）
    const blockScope = '只在这个 for 块内可访问';
    var badIdea = 'var 会泄漏到 demo 函数';
  }

  console.log(badIdea);    // 'var 会泄漏到 demo 函数'
  console.log(blockScope); // ❌ ReferenceError
}
```

#### 2.3 词法作用域（Lexical Scope）

词法作用域的意思是：**作用域在代码写下来时就确定了，与函数在哪里调用无关。**

```js
const name = 'global';

function outer() {
  const name = 'outer';

  function inner() {
    console.log(name); // 输出什么？
  }

  return inner;
}

const fn = outer();
fn(); // 'outer' —— 而非 'global'
// inner 定义在 outer 内部，所以它的词法作用域就是 outer。
// 无论 fn 在哪里被调用，它始终查找 outer 内部的 name。
```

### 3. 闭包（Closure）

#### 3.1 闭包是什么

当一个函数引用了外部函数作用域中的变量，并且这个函数被传递到了外部函数之外去执行，就形成了闭包。

```js
function createCounter() {
  let count = 0;        // count 是 createCounter 的内部变量
                        // 正常情况下函数执行完 count 就该被回收了

  return function() {   // 返回的函数引用了 count
    count++;
    return count;
  };
}

const counter = createCounter();
console.log(counter()); // 1  —— count 还"活着"
console.log(counter()); // 2  —— 而且状态被保留了
console.log(counter()); // 3
// counter 函数持有对 count 的引用，导致 createCounter 的作用域无法被 GC 回收
```

#### 3.2 闭包的内存影响

```js
// ❌ 循环中不小心创建闭包可能导致内存泄漏
function createButtons() {
  const largeData = new Array(1000000).fill('*');

  for (let i = 0; i < 10; i++) {
    // 每个回调都引用了 largeData，导致无法释放
    document.getElementById('btn' + i).onclick = function() {
      console.log(largeData.length); // 闭包引用
    };
  }
}

// ✅ 改进：只引用需要的数据
function createButtonsBetter() {
  const length = new Array(1000000).fill('*').length;

  for (let i = 0; i < 10; i++) {
    document.getElementById('btn' + i).onclick = function() {
      console.log(length); // 只引用 length 值
    };
  }
}
```

> **避坑提示**：闭包本身不会导致内存泄漏，但不正确的使用（如引用了不需要的大对象）会增加内存占用。不需要时手动解除引用（赋 `null`）。

#### 3.3 闭包的实战应用

**场景1：数据私有化**

```js
function createPerson(name) {
  // 私有变量，外部无法直接访问
  let age = 0;

  return {
    getName() { return name; },
    getAge() { return age; },
    birthday() { age++; },
    // 没有 setAge —— age 只能通过 birthday 递增
  };
}

const p = createPerson('Alice');
console.log(p.age);      // undefined —— 无法直接读取
console.log(p.getAge()); // 0
p.birthday();
console.log(p.getAge()); // 1
```

**场景2：函数柯里化（Currying）**

```js
// 柯里化：把多参数函数变成单参数函数链
const add = a => b => a + b;

const add5 = add(5);
console.log(add5(3));  // 8
console.log(add5(10)); // 15
```

**场景3：防抖（Debounce）**

```js
function debounce(fn, delay) {
  let timer = null;            // 闭包保存 timer

  return function(...args) {
    clearTimeout(timer);       // 清除上一次的定时器
    timer = setTimeout(() => {
      fn.apply(this, args);    // 延迟执行
    }, delay);
  };
}

// 使用：搜索框输入后 300ms 才发起请求
const searchInput = document.getElementById('search');
searchInput.addEventListener('input', debounce(function(e) {
  console.log('搜索:', e.target.value);
}, 300));
```

**场景4：节流（Throttle）**

```js
function throttle(fn, interval) {
  let lastTime = 0;           // 闭包保存上一次执行时间

  return function(...args) {
    const now = Date.now();
    if (now - lastTime >= interval) {
      fn.apply(this, args);
      lastTime = now;
    }
  };
}

// 使用：滚动事件每 200ms 最多触发一次
window.addEventListener('scroll', throttle(function() {
  console.log('scroll 事件触发');
}, 200));
```

### 4. IIFE（立即执行函数表达式）

```js
// IIFE：定义后立即执行，创建一个独立作用域
(function() {
  const privateVar = '只有我能看到';
  console.log('IIFE 执行了！');
})();

// 箭头函数版
(() => {
  console.log('箭头函数 IIFE');
})();

// 比 IIFE 更现代的替代方案：块级作用域
{
  const privateVar = '块级作用域';
  console.log('不需要 IIFE 了');
}
// 但 IIFE 有个块级作用域做不到的事：返回一个值
const result = (() => {
  const x = 10;
  const y = 20;
  return x + y;
})();
console.log(result); // 30
```

### 5. 参数默认值、剩余参数、展开运算符

#### 5.1 参数默认值

```js
// 旧写法
function greet(name) {
  name = name || '匿名';
  console.log('你好，' + name);
}

// ES6 默认参数（更安全）
function greet(name = '匿名') {
  console.log(`你好，${name}`);
}

greet();        // '你好，匿名'
greet('Alice'); // '你好，Alice'
greet('');      // '你好，' —— 不会像 || 那样把空字符串当 falsy
greet(0);       // '你好，0' —— 同理

// 默认值可以使用前面的参数
function createUser(name, role = name === 'admin' ? '管理员' : '普通用户') {
  return { name, role };
}
console.log(createUser('admin')); // { name: 'admin', role: '管理员' }
```

#### 5.2 剩余参数（Rest Parameters）

```js
// ...rest 收集剩余参数为数组，必须放在最后
function sum(...numbers) {
  return numbers.reduce((total, n) => total + n, 0);
}

console.log(sum(1, 2, 3, 4, 5)); // 15

// 解构中的剩余
const [first, second, ...others] = [10, 20, 30, 40, 50];
console.log(first);  // 10
console.log(second); // 20
console.log(others); // [30, 40, 50]
```

#### 5.3 展开运算符（Spread Operator）

```js
// 数组展开
const arr1 = [1, 2];
const arr2 = [3, 4];
const merged = [...arr1, ...arr2]; // [1, 2, 3, 4]
const copy = [...arr1];            // 浅拷贝

// 对象展开
const defaults = { theme: 'light', lang: 'zh' };
const userSettings = { lang: 'en' };
const final = { ...defaults, ...userSettings };
console.log(final); // { theme: 'light', lang: 'en' } —— 后者覆盖前者

// 函数调用展开
const nums = [5, 10, 15];
console.log(Math.max(...nums));          // 15，等价于 Math.max(5, 10, 15)
```

## 常见问题

### Q1：箭头函数什么时候不能用？

以下场景不能用箭头函数：
1. **对象方法**：`{ fn: () => {} }`，因为 `this` 不会指向对象
2. **构造函数**：`new Arrow()` 会报错，箭头函数没有 `[[Construct]]` 内部方法
3. **需要 `arguments` 对象**：箭头函数没有 `arguments`，要用剩余参数替代
4. **绑定事件处理器且需要通过 `this` 访问 DOM 元素**（用 `e.currentTarget` 替代）

### Q2：闭包会导致内存泄漏吗？

闭包本身不会。但如果你在闭包中保留了不需要的大对象引用，又不做清理，就会导致内存占用过高。合理使用闭包是没问题的，不需要时赋 `null` 即可。

### Q3：`arguments` 和剩余参数的区别？

| 特性 | arguments | 剩余参数 |
|------|-----------|----------|
| 类型 | 类数组对象（不是真数组） | 真正的数组 |
| 箭头函数 | ❌ 不可用 | ✅ 可用 |
| 命名 | 只能用在函数体 | 可自定义参数名 |

### Q4：函数声明和函数表达式该选哪个？

默认用函数声明（可读性好、有提升）。需要给回调传匿名函数时用函数表达式或箭头函数。需要绑定 `this` 时根据场景选择。

### Q5：为什么 `var` 在 for 循环中会导致闭包 bug？

```js
// ❌ 经典 bug：所有回调打印的都是 3
for (var i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 100);
}
// 输出：3, 3, 3 —— var 是函数作用域，所有回调共享同一个 i

// ✅ 修复方案1：用 let（块级作用域）
for (let i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 100);
}
// 输出：0, 1, 2

// ✅ 修复方案2：旧时代的 IIFE 折衷
for (var i = 0; i < 3; i++) {
  ((j) => setTimeout(() => console.log(j), 100))(i);
}
```

### Q6：展开运算符是深拷贝还是浅拷贝？

浅拷贝。只复制第一层，嵌套对象仍是引用：

```js
const original = [{ a: 1 }, { b: 2 }];
const copy = [...original];
copy[0].a = 999;
console.log(original[0].a); // 999 —— 被改了！
```

### Q7：为什么 `typeof fn === 'function'` 能判断函数？

函数是 Object 的子类型，但它是"可调用对象"，内部有 `[[Call]]` 方法。JavaScript 特殊处理了 `typeof` 使其对函数返回 `'function'`。

### Q8：IIFE 前面为什么要加分号？

```js
// 不带分号
const a = 1
(function() {})() // 被解析成 const a = 1(function() {})() —— 类型错误
```

JS 自动分号插入（ASI）不会在 `()` 之前插入分号，所以前面的分号是必要的防御。用 `void` 或者前置 `;` 也是常见写法：

```js
;(function() { console.log('safe'); })()
```

## 实操作业

### 🟢 基础：函数定义改写

将以下函数声明改写为函数表达式和箭头函数：

```js
function multiply(a, b) {
  return a * b;
}

function greet(name) {
  const greeting = 'Hello, ' + name;
  return greeting;
}
```

### 🟡 进阶：实现 once 函数

实现 `once(fn)`，确保传入的函数只会被执行一次，后续调用返回第一次的结果：

```js
function once(fn) {
  // 你的代码
}

const init = once(() => {
  console.log('初始化...');
  return { ready: true };
});

console.log(init()); // 打印 '初始化...'，返回 { ready: true }
console.log(init()); // 不打印，直接返回 { ready: true }
console.log(init()); // 同上
```

### 🔴 挑战：实现通用的 compose 函数

```js
// compose(f, g, h)(x) 等价于 f(g(h(x)))——从右到左执行
function compose(...fns) {
  // 你的代码：返回一个组合函数
}

const add1 = x => x + 1;
const double = x => x * 2;
const square = x => x * x;

const fn = compose(add1, double, square);
console.log(fn(3));
// 执行顺序：square(3) → double(9) → add1(18) = 19
```

### 作业参考与验收

- **基础**：表达式 const multiply=function(a,b){return a*b}；箭头 const multiply=(a,b)=>a*b。调用声明可早于定义，表达式不能在初始化前调用；箭头没有自身 this/arguments。
- **进阶**：闭包保存 called 和 result；首次成功后后续返回同一 result，并用 apply 保留 this。明确首次抛错是否允许重试，异步场景保存 Promise 防止并发重复执行。
- **挑战**：返回 x=>fns.reduceRight((value,fn)=>fn(value),x)。compose()(x) 返回 x；多参数首函数或异步 compose 需要另定契约，不把同步一元版说成支持所有函数。

## 本节小结

- **三种函数定义**：声明（提升）、表达式（不提升）、箭头函数（无自己的 this）
- **作用域链**：从内向外查找，词法作用域在编写时就已确定
- **闭包 = 函数 + 其引用的外部变量**：是实现数据私有化、柯里化、防抖节流的核心机制
- **IIFE**：创建独立作用域的老方法，现代代码中块级作用域可替代大部分场景
- **剩余参数和展开运算符**：`...` 在不同位置语义不同，但本质都是"打散/收集"
